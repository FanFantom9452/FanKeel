```js
// `failed` is the name on every `not ok` line that is a leaf: node prints a
// parent's result after its children's, so a test point whose previous test
// point sat deeper is a parent. A `# TODO` or `# SKIP` does not fail the run.
// failedLeaves also reads the `location:` line in that point's yaml block —
// the file the test is in, with its :line:col dropped and the yaml's doubled
// backslashes undone — or null where there is none.
function failedLeaves(text) {
    const out = [];
    let prev = -1;
    let open = null;
    for (const line of String(text).split(/\r?\n/)) {
        const point = /^(\s*)(not )?ok \d+ - (.+)$/.exec(line);
        if (point) {
            const indent = point[1].length;
            open = null;
            if (point[2] && indent >= prev && !/\s#\s*(TODO|SKIP)\b/i.test(point[3])) {
                open = { name: point[3], file: null };
                out.push(open);
            }
            prev = indent;
            continue;
        }
        const loc = open && !open.file ? /^\s+location: '(.+?)(?::\d+:\d+)?'$/.exec(line) : null;
        if (loc) open.file = loc[1].replace(/\\\\/g, '\\');
    }
    return out;
}

function parseTap(text) {
    const pass = /^# pass (\d+)/m.exec(text);
    const fail = /^# fail (\d+)/m.exec(text);
    return { pass: pass ? Number(pass[1]) : null, fail: fail ? Number(fail[1]) : null, failed: failedLeaves(text).map((l) => l.name) };
}
```

```js
// test-16: a full run is red because of one test file — every failing leaf has
// a `location:` in the same file, that file is inside `tree`, and their number
// is the run's own `fail` (a cancelled file or a failing hook makes them
// differ, and then nothing is retried). { file (relative to tree), names } or null.
function retryTarget(result, tree) {
    if (result.ok || !Array.isArray(result.failed) || !result.failed.length) return null;
    const leaves = failedLeaves(result.stdout);
    if (leaves.length !== result.failed.length || leaves.length !== result.fail || leaves.some((l) => !l.file)) return null;
    const files = [...new Set(leaves.map((l) => path.resolve(l.file)))];
    if (files.length !== 1) return null;
    const rel = path.relative(tree, files[0]);
    if (!rel || rel.startsWith('..') || path.isAbsolute(rel)) return null;
    return { file: rel, names: leaves.map((l) => l.name) };
}

// test-16 (decided 2026-10-21: accept the relaxation, a real intermittent
// failure can pass this way): a red node --test run whose failures are all in
// one file is run once more on that file alone, with the same command, env and
// lock. Green alone, the run is recorded green, `retried` holding the test
// names and `retriedFile` the file; red alone, or two files red, the first
// result stands. `run` is runSuite; a test passes its own.
function retryOnce(tree, result, extraEnv, opts, run = runSuite) {
    const target = NODE_TEST.test(result.command) ? retryTarget(result, tree) : null;
    if (!target) return result;
    const again = run(tree, result.command + ' "' + target.file + '"', extraEnv, opts);
    if (!again.ok) return result;
    return { ...result, ok: true, retried: target.names, retriedFile: target.file.split(path.sep).join('/'), ms: result.ms + again.ms, stdout: '' };
}
```

## Task 8: 測 serve 的 SIGINT/SIGTERM 收尾
