# Main-thread Stop hook Implementation Plan

**Goal:** refuse, once, a main-session turn that ends while its fankeel stage is open and nothing in the background will wake the session.
**Architecture:** `lib/stop.js` reads the session record and the session transcript and returns a reason or null; `hooks/stop.js` is the thin Stop hook around it, registered in `.claude-plugin/plugin.json`. `stop_hook_active` makes the refusal one-shot, so a turn that really is waiting on the user ends on its second try.
**Tech Stack:** Node >=22.5 (`package.json:8`), `node:test`, no dependencies.
**Spec:** 2026-10-22-stop-hook-design.md

## Global Constraints

From `node scripts/map.js` (`.fankeel/map.md`), `package.json` and the tests; the repo has no `CLAUDE.md`.

- `package.json:8`: `"node": ">=22.5"`; `package.json` has no `dependencies` — add none.
- Tests are `node --test`; an implementer runs only its own test file.
- `lib/` never requires `scripts/` or `hooks/` (map tree: "nothing here reaches into scripts/ or hooks/").
- Every hook exits 0 on every path and prints nothing for a session with no active task (`lib/hook.js:18-23`, `hooks/agent-end.js:10-12`); use `run` and `parse` from `lib/hook.js`.
- `lib/` and `hooks/` indent 4 spaces; `tests/` indent 2 spaces.
- `tests/contract.test.js:300-340` counts the registered hooks against `docs/01-guide/development.md` ("all fourteen hooks", "The other seven are not load-bearing") and `tests/hook.test.js:3` ("all fourteen hooks"); its `WORDS` array ends at `'fourteen'`. `Stop` is not in its `BLOCKING` set.
- `tests/inventory.test.js:71` requires `hooks/*.js` to equal what `plugin.json` references.
- `tests/away-docs.test.js:21` asserts `docs/02-architecture/layout.md` still says "Six are not rowed below" — give `stop.js` its own row rather than adding it to that sentence.
- File length cap `READ_CAP` 1500, at most 3 `Modify:` per task (`FILE_CAP`).

## Risks

- The Stop payload in the desktop app may lack `transcript_path` — Task 1 — `entries` returns null for an unreadable path and `refusal` then lets the stop through.
- An Agent result wording other than `Async agent launched` would read as foreground and could refuse a stop wrongly, once — Task 1 — the reason text tells the model to end the turn again when it is waiting.

```js
'use strict';

// The main-thread Stop check behind hooks/stop.js. A controller that ends its
// turn on a status line while its stage is open, with nothing in the background
// that will notify the session, leaves it idle until the user types — the top
// cause in docs/90-agent/judgements/2026-10-22-stage-agent-pause.md. The rules
// schedule a wake-up for three cases only (lib/stages.js AWAIT_RULE); this is
// the catch for the rest.

const fs = require('node:fs');

const ASYNC_AGENT = /Async agent launched/;
const NOTIFIED = /<tool-use-id>([^<]+)<\/tool-use-id>/g;

// Every line of the session transcript, or null when it cannot be read — a
// payload with no transcript is a stop let through, never one refused blind.
function entries(transcriptPath) {
    let raw;
    try {
        raw = fs.readFileSync(transcriptPath, 'utf8');
    } catch (e) {
        return null;
    }
    const out = [];
    for (const line of raw.split('\n')) {
        if (!line) continue;
        try {
            out.push(JSON.parse(line));
        } catch (e) {
            // a half-written last line
        }
    }
    return out;
}

function blocks(entry) {
    const c = entry && entry.message && entry.message.content;
    if (typeof c === 'string') return [{ type: 'text', text: c }];
    return Array.isArray(c) ? c : [];
}

function textOf(content) {
    if (typeof content === 'string') return content;
    if (Array.isArray(content)) return content.map((b) => (b && typeof b.text === 'string' ? b.text : '')).join('\n');
    return '';
}

// Work the harness will report on: a Bash or PowerShell call run in the
// background, or an Agent whose result says it launched async — each pending
// until a <task-notification> names its tool-use-id — and a SendMessage, which
// restarts an agent, pending until any notification follows it.
function pendingWork(list) {
    const launched = new Set();
    const agents = new Set();
    let sentAt = -1;
    let notifiedAt = -1;
    list.forEach((entry, i) => {
        if (!entry || entry.isSidechain) return;
        for (const b of blocks(entry)) {
            if (!b || typeof b !== 'object') continue;
            if (entry.type === 'assistant' && b.type === 'tool_use') {
                const input = b.input || {};
                if ((b.name === 'Bash' || b.name === 'PowerShell') && input.run_in_background === true) launched.add(b.id);
                else if (b.name === 'Agent') agents.add(b.id);
                else if (b.name === 'SendMessage') sentAt = i;
            } else if (entry.type === 'user' && b.type === 'tool_result') {
                if (agents.has(b.tool_use_id) && ASYNC_AGENT.test(textOf(b.content))) launched.add(b.tool_use_id);
            } else if (entry.type === 'user' && b.type === 'text' && b.text.includes('<task-notification>')) {
                for (const m of b.text.matchAll(NOTIFIED)) launched.delete(m[1].trim());
                notifiedAt = i;
            }
        }
    });
    return launched.size > 0 || sentAt > notifiedAt;
}

// The reason to refuse this stop, or null. No active task, no stage, a user
// marked away (hooks/gate.js held the question and this turn is meant to end),
// an unreadable transcript or work still pending all let it through.
function refusal(data, transcriptPath) {
    if (!data || data.active !== true || !data.stage) return null;
    if (typeof data.awayAt === 'number') return null;
    const list = entries(transcriptPath);
    if (!list || pendingWork(list)) return null;
    return 'fankeel: ' + data.stage + ' is open and nothing in the background will wake this session.'
        + ' Do the next step now: dispatch it (then run await.js in the background), or ask the stage\'s gate with AskUserQuestion.'
        + ' If you are answering the user or waiting on them, end the turn again; this is asked once.';
}

module.exports = { entries, pendingWork, refusal };
```

```js
// hooks/stop.js end to end: a real Stop payload against a project with a
// session record and a transcript on disk.
const test = require('node:test');
const assert = require('node:assert/strict');
const fs = require('node:fs');
const path = require('node:path');
const { execFileSync } = require('node:child_process');
const mkTmp = require('./tmp.js');

const HOOK = path.join(__dirname, '..', 'hooks', 'stop.js');
const SESSION = 'aaaaaaaa-0000-4000-8000-0000000000f1';

function project(active) {
  const root = mkTmp('fankeel-stop-hook-');
  fs.mkdirSync(path.join(root, '.fankeel', 'sessions'), { recursive: true });
  fs.writeFileSync(path.join(root, '.fankeel', 'sessions', SESSION + '.json'), JSON.stringify({
    task: 't', claims: [], stage: 'build', active, started: '2026-10-22T00:00:00.000Z', updated: new Date().toISOString(),
  }));
  const transcript = path.join(root, 'transcript.jsonl');
  fs.writeFileSync(transcript, JSON.stringify({ type: 'assistant', message: { role: 'assistant', content: [{ type: 'text', text: '等 station 遠端 session 收工' }] } }) + '\n');
  return { root, transcript };
}

function fire({ root, transcript }, extra) {
  return execFileSync(process.execPath, [HOOK], {
    input: JSON.stringify(Object.assign({ session_id: SESSION, cwd: root, hook_event_name: 'Stop', stop_hook_active: false, transcript_path: transcript }, extra)),
    encoding: 'utf8',
    env: Object.assign({}, process.env, { CLAUDE_PROJECT_DIR: root, CLAUDE_CONFIG_DIR: mkTmp('fankeel-cfg-') }),
  });
}

test('a stop with the build stage open and nothing pending is refused', () => {
  const out = JSON.parse(fire(project(true)));
  assert.equal(out.decision, 'block');
  assert.match(out.reason, /build is open/);
});

test('a stop that already followed a refusal goes through', () => {
  assert.equal(fire(project(true), { stop_hook_active: true }), '');
});

test('a session with no active task prints nothing', () => {
  assert.equal(fire(project(false)), '');
});

test('the manifest runs stop.js on Stop', () => {
  const manifest = JSON.parse(fs.readFileSync(path.join(__dirname, '..', '.claude-plugin', 'plugin.json'), 'utf8'));
  const commands = manifest.hooks.Stop.flatMap((m) => m.hooks.map((h) => h.command));
  assert.ok(commands.some((c) => c.includes('hooks/stop.js')));
});
```

```js
#!/usr/bin/env node
'use strict';

// Stop, main thread only: a controller that ends its turn while its stage is
// open and nothing in the background will wake the session is refused once,
// with lib/stop.js's reason. stop_hook_active says this stop already followed
// a refusal, so it passes — a turn really waiting on the user ends on its
// second try.
//
// Same rules as every hook here: exit 0 on every path, and nothing at all for
// a session that owns no active task. One thing goes to stdout: the refusal.

const registry = require('../lib/registry.js');
const { refusal } = require('../lib/stop.js');
const { run, parse } = require('../lib/hook.js');

function main(raw) {
    const payload = parse(raw);
    if (!payload || payload.stop_hook_active === true) return;
    if (payload.hook_event_name !== undefined && payload.hook_event_name !== 'Stop') return;
    const root = registry.rootFor(payload);
    const mine = registry.readSession(root, payload.session_id);
    const reason = refusal(mine, payload.transcript_path);
    if (reason) process.stdout.write(JSON.stringify({ decision: 'block', reason }));
}

run(main);
```
