```js
// gate-19 (docs/90-agent/todo/gate-19.md): a stage gate reads the same at every
// stage — the move and the task at the head of the question, and a verb on the
// first two options. Only the first question: a patrol's per-entry questions
// after it keep their own. The all-capitals form was retired on 2026-10-30; it
// is refused so there is one format.
const stageWord = (name) => {
    const s = String(name || '');
    return s.charAt(0).toUpperCase() + s.slice(1).toLowerCase();
};
const STAGE_WORDS = ['survey', 'design', 'plan', 'build', 'verify', 'audit', 'land'].map(stageWord);
const WRAP_UP = 'Wrap up — 收工';

// The header was the finished stage alone until 2026-10-31, when the user asked
// for the move: the Claude desktop sidebar shows a waiting question's header
// beside "Waiting", and `Survey` there did not read as the question's own
// `Survey -> Plan`. So the header is the question's move with `→` between the
// stages and, past AskUserQuestion's twelve columns (`Survey→Design` is 13),
// the destination alone — the user chose `→Design` over abbreviating stages.
const ARROW = '→';
function gateHeader(from, to) {
    const full = stageWord(from) + ARROW + stageWord(to);
    return width(full) <= MAX_HEADER_WIDTH ? full : ARROW + stageWord(to);
}

function formatProblem(gate, next, route) {
    const q = gate.questions[0];
    const at = 'questions[0].';
    const header = String(q.header || '').trim();
    const m = /^([A-Z][a-z]+) -> ([A-Z][a-z]+|收工) \| [^:]+ : \S/.exec(String(q.question || ''));
    if (!m || !STAGE_WORDS.includes(m[1])) {
        return { at: at + 'question', detail: 'question must open "<This> -> <Next> | <project> : <task>", <This> the finished stage with only its first letter capital — one of ' + STAGE_WORDS.join(', ') };
    }
    const from = m[1];
    const want = gateHeader(from, m[2]);
    if (header !== want) {
        return { at: at + 'header', detail: 'header "' + q.header + '" must be "' + want + '": the question\'s move with ' + ARROW + ' between its stages, or ' + ARROW + '<Next> alone past ' + MAX_HEADER_WIDTH + ' columns' };
    }
```
