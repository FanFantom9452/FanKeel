# Pane and band redesign Implementation Plan

**Goal:** The two rows above the prompt answer only "which station, how much quota is left", the Pane answers "the task's detail and whether to act now", and a clash counts only sessions that are still alive.
**Architecture:** A new engine-free module `hooks/pane-bar.ts` holds what both drawings share: cell width, the ten-cell eighths bar, TokenBar's ramp as hex and the ctx / 5h / 7d colour rules. `hooks/pane-band.ts` and `hooks/pane-status.ts` stay pure functions from data to coloured segments, tested under plain node; `hooks/pane.ts` is the only file that asks the engine, and it hands them the token count, `rateLimits`, the plan's tasks and the live list `hooks/inject.js` writes on every prompt. Three patrol items in `lib/context.js` and two test files ride along as their own tasks.
**Tech Stack:** Node v24.21.0 (`node --test`; `.ts` files load by type stripping, so erasable TypeScript only), no dependencies, the Claude Code mod API typed in `.claude-plugin/types/claude-code/index.d.ts`.
**Spec:** 2026-10-12-pane-band-design.md

```js
// One transcript row, or null: the tail's first line is cut mid-row, and a row
// that does not parse carries nothing this file reads.
function rowOf(line) {
    try {
        const row = JSON.parse(line);
        return row && typeof row === 'object' ? row : null;
    } catch (e) {
        return null;
    }
}

const count = (v) => (typeof v === 'number' && Number.isFinite(v) ? v : 0);

// What the session currently holds, from the most recent usage record. The three
// input figures together are the context that was sent, whatever share of it the
// cache served. Only an assistant row's own `message.usage` is one: the same
// text inside a tool result or a message body is a quotation, and reading it
// reported 652k in play for a subagent that had read a test fixture.
function usedFrom(text) {
    let last = null;
    for (const line of text.split('\n')) {
        if (!line.includes('"usage"')) continue;
        const row = rowOf(line);
        const usage = row && row.type === 'assistant' && row.message && row.message.usage;
        if (!usage || typeof usage !== 'object') continue;
        last = count(usage.input_tokens) + count(usage.cache_creation_input_tokens) + count(usage.cache_read_input_tokens);
    }
    return last !== null && last > 0 ? last : null;
}

// Only a compact_boundary row records a compaction. The pattern still runs over
// the row's own text, because `compactMetadata` sits inside a string there.
function droppedFrom(text) {
    let last = null;
    for (const line of text.split('\n')) {
        if (!line.includes('compact_boundary')) continue;
        const row = rowOf(line);
        if (!row || row.type !== 'system' || row.subtype !== 'compact_boundary') continue;
        DROPPED.lastIndex = 0;
        let m;
        while ((m = DROPPED.exec(line)) !== null) last = parseInt(m[1], 10);
    }
    return Number.isFinite(last) ? last : null;
}
```

```ts
// `mark` is the warning sign drawn after the number, or ''.
export type Paint = { color: string; bold: boolean; mark: string }

export const WIDE = /[ᄀ-ᅟ⺀-꓏가-힣豈-﫿︰-﹏＀-｠￠-￦]/
// 💥 and 💀 are here. ⚠ (U+26A0) is not: it is counted one cell.

export const list = (v: unknown): string[] => (Array.isArray(v) ? v.filter((x): x is string => typeof x === 'string') : [])

export function width(s: string): number {
  let n = 0
  for (const ch of s) n += WIDE.test(ch) || EMOJI.test(ch) ? 2 : 1
  return n
}

export function cut(s: string, cols: number): string {
  if (width(s) <= cols) return s
  let out = ''
  for (const ch of s) {
    if (width(out + ch) > cols - 1) break
    out += ch
  }
  return out + '…'
}

// [from, xterm, bold]: the last step whose `from` is at or under the value wins.
type Step = [number, number, boolean?]
// Percent of the compact threshold: TokenBar's tier for a window of 500k or less.
const CTX: Step[] = [[0, 46], [8, 83], [16, 120], [24, 157], [32, 194], [36, 190], [40, 226], [45, 220], [50, 214], [55, 208],
  [60, 196, true], [65, 198, true], [70, 199, true], [75, 200, true], [80, 201, true]]
// Percent of the five-hour window used.
const FIVE: Step[] = [[0, 46], [10, 83], [20, 120], [30, 157], [40, 194], [45, 193], [50, 192], [55, 190], [60, 226], [65, 220], [70, 214], [75, 208],
  [80, 196, true], [85, 197, true], [90, 199, true], [95, 201, true]]

const DEV: Step[] = [[-999, 46], [-10, 83], [-7, 120], [-3, 157], [0, 194], [1, 193], [2, 192], [3, 191], [4, 190], [5, 226], [6, 214], [7, 202],

function step(steps: Step[], value: number): { color: string; bold: boolean } {
  let hit = steps[0]
  for (const s of steps) if (s[0] <= value) hit = s
  return { color: XTERM[hit[1]], bold: hit[2] === true }
}

export function ctxPaint(tokens: number, at: number): Paint {
  const percent = at > 0 ? (tokens * 100) / at : 0
  return { ...step(CTX, percent), mark: percent >= 80 ? '⚠' : '' }
}

export function fivePaint(used: number): Paint {
  return { ...step(FIVE, used), mark: used >= 95 ? '⚠' : '' }
}

const devMark = (dev: number): string => (dev >= 14 ? '💀' : dev >= 10 ? '💥' : dev >= 7 ? '⚠' : '')

// The deviation alone colours the seven-day meter, whatever was used; from 95
// percent used it is purple either way. With no deviation (no `resetsAt`) the
// colour is the five-hour rule on percent used, and there is no mark.
export function weekPaint(used: number, dev: number | null): Paint {
  if (dev === null) return { ...step(FIVE, used), mark: '' }
  if (used >= 95) return { color: XTERM[201], bold: true, mark: devMark(dev) }
  return { ...step(DEV, dev), mark: devMark(dev) }
}

const PART = ['', '▏', '▎', '▍', '▌', '▋', '▊', '▉']

// `cells` cells of bar: `fill` is the used part in eighths of a cell, rounded
// down, and `track` the rest. Either may be ''.
export function bar(part: number, whole: number, cells: number): { fill: string; track: string } {
  const eighths = whole > 0 ? Math.floor((Math.max(0, Math.min(part, whole)) * cells * 8) / whole) : 0
  const full = Math.floor(eighths / 8)
  const last = PART[eighths % 8]
  return { fill: '█'.repeat(full) + last, track: '░'.repeat(cells - full - (last ? 1 : 0)) }
}

// The instant the window resets, epoch milliseconds, or null.
export function resetMs(limit: Limit): number | null {
  const at = typeof limit.resetsAt === 'string' ? Date.parse(limit.resetsAt) : NaN
  return Number.isFinite(at) ? at : null
}

const WEEK = 7 * 24 * 3600 * 1000

// Percent used less the share of the seven days already gone by, in whole
// points, rounded half away from zero as TokenBar does. Null with no `resetsAt`.
export function deviation(limit: Limit, now: number): number | null {
  const reset = resetMs(limit)
  if (reset === null) return null
  const elapsed = Math.max(0, Math.min(1, 1 - (reset - now) / WEEK))
  const raw = limit.percentUsed - elapsed * 100
  // `|| 0`: never -0.
  return (raw >= 0 ? Math.floor(raw + 0.5) : Math.ceil(raw - 0.5)) || 0
}

// `1h 46m`, or `2d 12h 30m` where `days` is asked for; `now` once it is past.
export function countdown(ms: number, days: boolean): string {
  if (!(ms > 0)) return 'now'
  const total = Math.floor(ms / 60000)
  const two = (n: number): string => String(n).padStart(2, '0')
  const hours = Math.floor(total / 60)
  if (days) return `${Math.floor(hours / 24)}d ${two(hours % 24)}h ${two(total % 60)}m`
  return `${hours}h ${two(total % 60)}m`
}

// ` 66%`: three cells of number, so a column of them lines up.
export const pct = (n: number): string => `${String(Math.round(n)).padStart(3)}%`
export const kilo = (n: number): string => `${Math.round(n / 1000)}k`
export const signed = (dev: number): string => `${dev > 0 ? '+' : ''}${dev}%`

// The window of that kind (`five_hour`, `seven_day`), or null; one with no number is no window.
export function limitOf(limits: Limit[], kind: string): Limit | null {
  return limits.find((l) => !!l && l.kind === kind && typeof l.percentUsed === 'number' && Number.isFinite(l.percentUsed)) ?? null
}
```

```ts
// A row cut to `cols` cells: the segment that crosses the edge is cut, and the ones after it go.
function fit(r: Row, cols: number): Row {
  let left = cols
  const segs: Seg[] = []
  for (const seg of r.segs) {
    if (left <= 0) break
    const w = width(seg.text)
    segs.push(w <= left ? seg : { ...seg, text: cut(seg.text, left) })
    left -= w
  }
  return { ...r, segs }
}
```

```ts
// complete, ascending, or null when its first line names another plan.
export function ledgerDoneIds(text: string, claim: string): number[] | null {
  const lines = text.split(/\r?\n/)
  if (lines[0] !== LEDGER + claim) return null

// lib/plantasks.js parsePlan: the `## Task N:` headings of a plan, a fenced one
// not counted, each with whether its first `**Dispatch:**` line names the user.
// Any other `## ` heading ends the tasks.
export function planTasks(text: string): PlanTask[] {
  const tasks: PlanTask[] = []
  let task: PlanTask | null = null
  let said = false
  let fence = 0
  for (const raw of text.split(/\r?\n/)) {
    const line = raw.trim()
    const t = fence ? null : /^##\s+Task\s+(\d+):\s*(.*)$/.exec(line)
    if (t) {
      task = { n: Number(t[1]), name: t[2].trim(), user: false }
      said = false
      tasks.push(task)
      continue
    }
    if (!fence && /^##\s/.test(line)) task = null
    const f = /^`{3,}/.exec(line)
    if (f) {
      if (fence === 0) fence = f[0].length
      else if (f[0].length >= fence) fence = 0
      continue
    }
    if (fence || !task || said) continue
    const d = /^\*\*Dispatch:\*\*\s*(.*)$/.exec(line)
    if (!d) continue
    said = true
    const word = /^([a-z][a-z-]*)/i.exec(d[1])
    task.user = !!word && word[1].toLowerCase() === 'user'
  }
  return tasks
}
```

```ts
// The percent of the threshold the block says what a compact keeps at, and stops being dropped at.
const KEEP = 80
// Integer comparison, so a threshold needs no rounding.
const near = (context: Context): boolean => context.tokens * 100 >= KEEP * context.at

// The count and what is left before the automatic compact, then a bar on the
// scale 0 to the threshold (hooks/compact.ts HARD or FANKEEL_COMPACT_AT, never
// the model's window): ten to eighteen cells, as many as the label leaves.
function contextBlock(context: Context, inner: number): Row[] {
  const paint = ctxPaint(context.tokens, context.at)
  const hot = context.tokens >= context.at
  const label = hot ? ` 已到 ${kilo(context.at)}，compact 中` : ` 到 ${kilo(context.at)} 自動 compact`
  const { fill, track } = bar(context.tokens, context.at, Math.max(10, Math.min(18, inner - width(label))))
  const rows = [
    row('context', title('context', 'context'), inked(`  ${kilo(context.tokens)}`, paint), inked(paint.mark ? ` ${paint.mark}` : '', paint),
      dim(' · 還 '), { text: kilo(Math.max(0, context.at - context.tokens)) }, dim(' 可用')),
    row('context', inked(fill, paint), { text: track, color: DIM }, hot ? { text: label, color: GUARD, bold: true } : dim(label)),
  ]
  if (near(context)) rows.push(row('context', dim(cut(`留 ${KEPT.join(' · ')}`, inner))))
  return rows
}

// One row a window the engine gave: a ten-cell bar, percent used, what is left
// and when it resets. A row that does not fit drops the reset, then what is
// left. Under the seven-day row, its deviation from an even pace in a sentence.
function quotaBlock(limits: Limit[], inner: number, now: number): Row[] {
  const five = limitOf(limits, 'five_hour')
  const week = limitOf(limits, 'seven_day')
  if (!five && !week) return []
  const cells = (segs: Seg[]): number => segs.reduce((n, seg) => n + width(seg.text), 0)
  const meter = (label: string, limit: Limit, paint: Paint, mark: string, days: boolean): Row => {
    const { fill, track } = bar(limit.percentUsed, 100, 10)
    const reset = resetMs(limit)
    const base: Seg[] = [dim(label), inked(fill, paint), { text: track, color: DIM }, inked(` ${pct(limit.percentUsed)}`, paint), inked(mark ? ` ${mark}` : '', paint)]
    const left: Seg[] = [dim('  剩'), { text: pct(100 - limit.percentUsed) }]
    const until: Seg[] = reset === null ? [] : [dim(`  ↻ ${countdown(reset - now, days)}`)]
    const shape = [[...base, ...left, ...until], [...base, ...left]].find((segs) => cells(segs) <= inner) ?? base
    return row('quota', ...shape)
  }
  const rows = [row('quota', title('quota', '用量額度'))]
  if (five) {
    const paint = fivePaint(five.percentUsed)
    rows.push(meter('5h ', five, paint, paint.mark, false))
  }
  if (week) {
    const dev = deviation(week, now)
    const paint = weekPaint(week.percentUsed, dev)
    rows.push(meter('7d ', week, paint, '', true))
    if (dev !== null) {
      rows.push(row('quota', { text: '   ' }, inked(signed(dev), paint), inked(paint.mark ? ` ${paint.mark}` : '', paint),
        dim(`  ${dev > 0 ? '超前' : '落後'}平均進度`), { text: ` ${Math.abs(dev)} ` }, dim('點')))
    }
  }
  return rows
}
```

```ts
// `threshold`, written again for the same reason as findEntry — and the rate
// limits the same call reports: `five_hour` and `seven_day` are the two drawn.
// No context while the engine has no count; neither when it cannot be asked.
async function contextOf($: EngineInterface, logged: Logged): Promise<{ context: Context | null; limits: Limit[] }> {
  try {
    const usage = await $.session.usage()
    const limits = Array.isArray(usage.rateLimits) ? (usage.rateLimits as Limit[]) : []
    if (typeof usage.context.tokens !== 'number') return { context: null, limits }
    const raw = Number(await $.env.get('FANKEEL_COMPACT_AT'))
    return { context: { tokens: usage.context.tokens, at: Number.isFinite(raw) && raw > 0 ? raw : HARD }, limits }
  } catch (err) {
    logOnce($, logged, 'context', `fankeel pane: no token count, ${String(err)}`)
    return { context: null, limits: [] }
  }
}

// The newest plan this task claimed that has a build ledger of its own: the
// ledger is where lib/ledger.js ledgerPath puts it, and ledgerDoneIds refuses
// one whose first line names another plan. The tasks are the plan's own headings.
async function planOf($: EngineInterface, root: string, entry: Entry, cache: Cache, logged: Logged): Promise<PlanInfo | null> {
  const claims = Array.isArray(entry.claims) ? entry.claims : []
  for (let i = claims.length - 1; i >= 0; i--) {
    const claim = claims[i]
    if (typeof claim !== 'string' || !claim.endsWith('.md')) continue
    const name = claim.slice(Math.max(claim.lastIndexOf('/'), claim.lastIndexOf('\\')) + 1, -3)
    const file = `${root}/.fankeel/build/${name}/progress.md`
    if (!(await $.fs.exists(file))) continue
    const ledger = String(await $.fs.read(file))
    const done = ledgerDoneIds(ledger, claim)
    if (done === null) continue
    let tasks: PlanTask[] = []
    try {
      const planFile = `${root}/${claim}`
      const stat = await $.fs.stat(planFile)
      const stamp = `${stat.mtimeMs}:${stat.size}`
      let hit = cache.get(planFile)
      if (!hit || hit.stamp !== stamp) {
        hit = { stamp, value: planTasks(String(await $.fs.read(planFile))) }
        cache.set(planFile, hit)
      }
      tasks = hit.value as PlanTask[]
    } catch (err) {
      logOnce($, logged, 'plan', `fankeel pane: plan file unreadable, ${String(err)}`)
    }
    return tasks.length > 0 ? { name, done, tasks } : null
  }
  return null
}
```

```ts
// The neighbours hooks/inject.js found running at this session's last prompt
// (`.fankeel/build/live.json`), or null where that cannot be told: no file, one
// that does not parse, no `at`, or `ids` that is no list — lib/live.js writes
// null there when it could not tell. The mod has no pid to signal, so this file
// is all it knows of who is alive. Read every tick: it is one short line.
async function liveOf($: EngineInterface, root: string): Promise<{ at: number; ids: string[] } | null> {
  try {
    const file = `${root}/.fankeel/build/live.json`
    if (!(await $.fs.exists(file))) return null
    const got = JSON.parse(String(await $.fs.read(file))) as { at?: unknown; ids?: unknown } | null
    if (!got || typeof got.at !== 'number' || !Number.isFinite(got.at) || !Array.isArray(got.ids)) return null
    return { at: got.at, ids: got.ids.filter((x): x is string => typeof x === 'string') }
  } catch {
    return null
  }
}
```
