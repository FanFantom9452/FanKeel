# retro-all 匯出 ZIP 回傳 Implementation Plan

**Goal:** the person who runs `/fankeel-retro-all` sends the result back as one ZIP — the report, the session bundles it read and a whitelisted environment manifest — written by `record.js export`, and the receiver puts the bundles back with `record.js import`; nobody pushes.
**Architecture:** a new `lib/zip.js` writes stored-only ZIPs and reads stored and deflated ones, CRC-checked with the built-in `zlib.crc32`. `scripts/record.js` gains `export` (cut the cross report and each bundle's retro out of the month files, rewrite each bundle's home directory to `[HOME]` into a fresh gzip, build and redact the manifest, write `.fankeel/exports/retro-<date>-<sha8>.zip`) and `import` (verify CRC and sha256 for everything before writing anything, skip names already in `.fankeel/bundles/`, keep manifest and report under `.fankeel/imports/<stem>/`). The retro-all skill gets a gate option, a privacy question and a 收到 ZIP section.
**Tech Stack:** Node `>=22.5` (package.json `engines`; `zlib.crc32` landed in 22.2), CommonJS, `node:test` + `node:assert/strict`, no dependencies.
**Spec:** [2026-11-02-retro-zip-design.md](2026-11-02-retro-zip-design.md) — §1–§8.

## Risks

- `zlib.crc32` returns an unsigned 32-bit number and `readUInt32LE` reads one too — Task 1 — the round-trip test compares them directly; a signed mismatch fails it first.
- The `[HOME]` pattern is built from the real `os.homedir()` — Task 2 — the test runs record.js with `HOME` and `USERPROFILE` set to a made-up `zedtester` home, so the masking is checked without depending on the machine's own path; it also checks the bundle on disk still holds the original.
- `--cross` is already a boolean in `OPTIONS` (used by `retro --cross`) — Task 2 — `export` reads the anchor as the positional after the verb, so `record.js export --cross x-2026-11-02 --bundles …` parses with no new string flag; the refusal test passes the anchor the same way.
- `npm ls -g` is slow and may be missing — Task 2 — `globalPackages()` returns `'unavailable: <why>'` instead of failing, and only the `--global-packages` test calls it.
- The skill grows past its 3000-byte cap — Task 3 — it measures with `node scripts/input-check.js --skills` before raising the cap.

```js
const test = require('node:test');
const assert = require('node:assert/strict');
const zlib = require('node:zlib');
const zip = require('../lib/zip.js');

const BUNDLE = 'bundles/2026-11-02-aaaaaaaa.jsonl.gz';

// The same bytes with every `from` swapped for `to`; both are one length, so
// every offset in the ZIP still holds.
function rename(buf, from, to) {
    return Buffer.from(buf.toString('latin1').split(from).join(to), 'latin1');
}

// A ZIP the way a file manager writes one: a directory entry, then the file
// compressed with DEFLATE.
function deflated(name, data) {
    const parts = [];
    const central = [];
    let offset = 0;
    for (const [n, body, method] of [['bundles/', Buffer.alloc(0), 0], [name, data, 8]]) {
        const nb = Buffer.from(n, 'utf8');
        const packed = method === 8 ? zlib.deflateRawSync(body) : body;
        const lh = Buffer.alloc(30);
        lh.writeUInt32LE(0x04034b50, 0);
        lh.writeUInt16LE(20, 4);
        lh.writeUInt16LE(method, 8);
        lh.writeUInt32LE(zlib.crc32(body), 14);
        lh.writeUInt32LE(packed.length, 18);
        lh.writeUInt32LE(body.length, 22);
        lh.writeUInt16LE(nb.length, 26);
        const ch = Buffer.alloc(46);
        ch.writeUInt32LE(0x02014b50, 0);
        ch.writeUInt16LE(20, 4);
        ch.writeUInt16LE(20, 6);
        ch.writeUInt16LE(method, 10);
        ch.writeUInt32LE(zlib.crc32(body), 16);
        ch.writeUInt32LE(packed.length, 20);
        ch.writeUInt32LE(body.length, 24);
        ch.writeUInt16LE(nb.length, 28);
        ch.writeUInt32LE(offset, 42);
        parts.push(lh, nb, packed);
        central.push(ch, nb);
        offset += 30 + nb.length + packed.length;
    }
    const cd = Buffer.concat(central);
    const end = Buffer.alloc(22);
    end.writeUInt32LE(0x06054b50, 0);
    end.writeUInt16LE(2, 8);
    end.writeUInt16LE(2, 10);
    end.writeUInt32LE(cd.length, 12);
    end.writeUInt32LE(offset, 16);
    return Buffer.concat([...parts, cd, end]);
}

test('write then read gives back every entry, name and bytes', () => {
    const entries = [
        { name: 'manifest.json', data: Buffer.from('{"export":1}') },

    const buf = zip.write([{ name: 'report.md', data: Buffer.from('hello world') }]);
    buf[buf.indexOf('hello world')] = 'j'.charCodeAt(0);
    assert.throws(() => zip.read(buf), /CRC/);
});

test('a name that is absolute or climbs out with .. is refused on write and on read', () => {
    for (const name of ['../x', 'a/../../x', '/etc/x', 'C:/x', '\\x']) {
        assert.throws(() => zip.write([{ name, data: Buffer.from('x') }]), /absolute or climbs out/, name);
    }
    const buf = rename(zip.write([{ name: 'zz/x', data: Buffer.from('x') }]), 'zz/x', '../x');
    assert.throws(() => zip.read(buf), /absolute or climbs out/);
});

test('a ZIP compressed again with DEFLATE reads, its directory entry skipped', () => {
    const data = zlib.gzipSync('{"bundle":1}\n'.repeat(50));
    const back = zip.read(deflated(BUNDLE, data));
    assert.deepEqual(back.map((e) => e.name), [BUNDLE]);
    assert.ok(back[0].data.equals(data));
});

test('ZIP64, encryption and a buffer that is not a ZIP are refused by name', () => {
    const empty = zip.write([]);
    empty.writeUInt16LE(0xffff, 10);
    assert.throws(() => zip.read(empty), /ZIP64/);
    const one = zip.write([{ name: 'report.md', data: Buffer.from('x') }]);
    const cd = one.readUInt32LE(one.length - 6);
    one.writeUInt16LE(1, cd + 8);
    assert.throws(() => zip.read(one), /encrypted/);
    assert.throws(() => zip.read(Buffer.from('not a zip at all, just text')), /not a ZIP/);
});
```

```js
'use strict';

// The ZIP container a retro export travels in
// (docs/90-agent/plans/2026-11-02-retro-zip-design.md §6). `write` stores only,
// since every bundle is gzip'd already; `read` takes stored and deflated
// entries, so a ZIP the receiver's file manager compressed again still reads.
// No ZIP64, encryption or split archives: each is refused by name.

const zlib = require('node:zlib');

const LOCAL = 0x04034b50;
const CENTRAL = 0x02014b50;
const END = 0x06054b50;
const UTF8 = 0x0800;

// An entry name that would land outside the directory it is unpacked into.
function unsafe(name) {
    return !name || /^[\\/]/.test(name) || /^[A-Za-z]:/.test(name) || name.split(/[\\/]/).includes('..');
}

function dosTime(d) {
    return {
        time: (d.getHours() << 11) | (d.getMinutes() << 5) | Math.floor(d.getSeconds() / 2),
        date: ((Math.max(d.getFullYear(), 1980) - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate(),
    };
}

function write(entries, when) {
    if (entries.length > 0xfffe) throw new Error('more than 65534 entries needs ZIP64, which is not supported');
    const { time, date } = dosTime(when || new Date());
    const locals = [];
    const centrals = [];
    let offset = 0;
    for (const e of entries) {
        if (unsafe(e.name)) throw new Error('entry name ' + JSON.stringify(e.name) + ' is absolute or climbs out with ..');
        const name = Buffer.from(e.name, 'utf8');
        const data = Buffer.from(e.data);
        if (data.length >= 0xffffffff || offset >= 0xffffffff) throw new Error(e.name + ' is past 4 GB, which needs ZIP64');
        const crc = zlib.crc32(data);
        const lh = Buffer.alloc(30);
        lh.writeUInt32LE(LOCAL, 0);
        lh.writeUInt16LE(20, 4);
        lh.writeUInt16LE(UTF8, 6);
        lh.writeUInt16LE(0, 8);
        lh.writeUInt16LE(time, 10);
        lh.writeUInt16LE(date, 12);
        lh.writeUInt32LE(crc, 14);
        lh.writeUInt32LE(data.length, 18);
        lh.writeUInt32LE(data.length, 22);
        lh.writeUInt16LE(name.length, 26);
        lh.writeUInt16LE(0, 28);
        const ch = Buffer.alloc(46);
        ch.writeUInt32LE(CENTRAL, 0);
        ch.writeUInt16LE(20, 4);
        ch.writeUInt16LE(20, 6);
        ch.writeUInt16LE(UTF8, 8);
        ch.writeUInt16LE(0, 10);
        ch.writeUInt16LE(time, 12);
        ch.writeUInt16LE(date, 14);
        ch.writeUInt32LE(crc, 16);
        ch.writeUInt32LE(data.length, 20);
        ch.writeUInt32LE(data.length, 24);
        ch.writeUInt16LE(name.length, 28);
        ch.writeUInt32LE(offset, 42);
        locals.push(lh, name, data);
        centrals.push(ch, name);
        offset += lh.length + name.length + data.length;
    }
    const cd = Buffer.concat(centrals);
    const end = Buffer.alloc(22);
    end.writeUInt32LE(END, 0);
    end.writeUInt16LE(entries.length, 8);
    end.writeUInt16LE(entries.length, 10);
    end.writeUInt32LE(cd.length, 12);
    end.writeUInt32LE(offset, 16);
    return Buffer.concat([...locals, cd, end]);
}

// The end record is the last 22 bytes, or sits before a comment of up to 65535.
function endRecord(buf) {
    for (let at = buf.length - 22; at >= 0 && at >= buf.length - 22 - 0xffff; at--) {
        if (buf.readUInt32LE(at) === END) return at;
    }
    throw new Error('no end-of-central-directory record: not a ZIP, or cut short');
}

function read(buf) {
    const end = endRecord(buf);
    if (buf.readUInt16LE(end + 4) !== 0 || buf.readUInt16LE(end + 6) !== 0) throw new Error('a split (multi-volume) ZIP is not supported');
    const count = buf.readUInt16LE(end + 10);
    const cdOffset = buf.readUInt32LE(end + 16);
    if (count === 0xffff || cdOffset === 0xffffffff) throw new Error('a ZIP64 archive is not supported');
    const out = [];
    let at = cdOffset;
    for (let i = 0; i < count; i++) {
        if (at + 46 > buf.length || buf.readUInt32LE(at) !== CENTRAL) throw new Error('central directory entry ' + i + ' is damaged');
        const flags = buf.readUInt16LE(at + 8);
        const method = buf.readUInt16LE(at + 10);
        const crc = buf.readUInt32LE(at + 16);
        const csize = buf.readUInt32LE(at + 20);
        const usize = buf.readUInt32LE(at + 24);
        const nlen = buf.readUInt16LE(at + 28);
        const elen = buf.readUInt16LE(at + 30);
        const clen = buf.readUInt16LE(at + 32);
        const local = buf.readUInt32LE(at + 42);
        const name = buf.subarray(at + 46, at + 46 + nlen).toString('utf8');
        at += 46 + nlen + elen + clen;
        if (flags & 1) throw new Error(name + ' is encrypted, which is not supported');
        if (csize === 0xffffffff || usize === 0xffffffff || local === 0xffffffff) throw new Error(name + ' needs ZIP64, which is not supported');
        if (unsafe(name)) throw new Error('entry name ' + JSON.stringify(name) + ' is absolute or climbs out with ..');
        if (name.endsWith('/')) continue;
        if (local + 30 > buf.length || buf.readUInt32LE(local) !== LOCAL) throw new Error(name + ': its local header is missing');
        const from = local + 30 + buf.readUInt16LE(local + 26) + buf.readUInt16LE(local + 28);
        const raw = buf.subarray(from, from + csize);
        let data;
        if (method === 0) data = Buffer.from(raw);
        else if (method === 8) data = zlib.inflateRawSync(raw);
        else throw new Error(name + ' uses compression method ' + method + '; only stored and deflate are read');
        if (data.length !== usize || zlib.crc32(data) !== crc) throw new Error(name + ' fails its CRC check: the ZIP is damaged');
        out.push({ name, data });
    }
    return out;
}

module.exports = { write, read };
```

```js
const crypto = require('node:crypto');

const SCRIPT = path.join(__dirname, '..', 'scripts', 'record.js');
const WIN = process.platform === 'win32';
// os.homedir() reads USERPROFILE on Windows and HOME elsewhere; a made-up one
// makes "no home directory left" checkable on any machine.
const HOME = WIN ? 'C:\\Users\\zedtester' : '/home/zedtester';
const SPELLINGS = [HOME + (WIN ? '\\proj' : '/proj'), HOME.split('\\').join('/') + '/b'].concat(WIN ? ['/c/Users/zedtester/c', 'C:\\USERS\\ZEDTESTER\\d'] : []);

const REPORT = 'recurring: mover commits — 2/2 bundles — evidence: aaaaaaaa turn 3, bbbbbbbb turn 5';
const sha256 = (buf) => crypto.createHash('sha256').update(buf).digest('hex');

// A bundle the way lib/bundle.js packs one: the head line, then { f, e } rows.
function gz(id) {
    const head = JSON.stringify({ bundle: 1, sessions: [id], files: [id + '.jsonl'], redacted: {} });
    const rows = SPELLINGS.map((p) => JSON.stringify({ f: id + '.jsonl', e: { type: 'user', version: '2.1.0', cwd: SPELLINGS[0], message: { role: 'user', content: 'open ' + p } } }));
    return zlib.gzipSync(head + '\n' + rows.join('\n') + '\n');
}

function run(root, args, input) {
    try {
        return { out: execFileSync(process.execPath, [SCRIPT, ...args, '--root', root], { encoding: 'utf8', input: input || '', env: { ...process.env, HOME, USERPROFILE: HOME } }), code: 0 };
    } catch (e) {
        return { out: String(e.stdout || '') + String(e.stderr || ''), code: e.status };
    }
}

// Two bundles, a retro filed for the first, and a cross report; returns its anchor.
function seed() {
    const root = tmp('fankeel-rexp-');
    const dir = path.join(root, '.fankeel', 'bundles');
    fs.mkdirSync(dir, { recursive: true });
    fs.writeFileSync(path.join(dir, A), gz('aaaaaaaa'));
    fs.writeFileSync(path.join(dir, B), gz('bbbbbbbb'));

    assert.equal(r.code, 0, r.out);

}

const exported = (root, out) => path.join(root, ...(/^export: (\S+)$/m.exec(out)[1]).split('/'));

    const got = entries(exported(s.root, r.out));

    for (const b of m.bundles) {

        assert.equal(b.bytes, data.length, b.name);
```

```js
// The section `anchor` opens in any month file under `dir`, up to the next anchor.
function sectionIn(dir, anchor) {
    let names = [];
    try { names = fs.readdirSync(dir).filter((n) => n.endsWith('.md')).sort(); } catch (e) { return null; }
    for (const n of names) {
        const text = fs.readFileSync(path.join(dir, n), 'utf8');
        const at = text.indexOf('<a id="' + anchor + '"></a>');
        if (at < 0) continue;
        const next = text.indexOf('\n<a id="', at + 1);
        return text.slice(at, next < 0 ? text.length : next + 1).replace(/\n*$/, '\n');
    }
    return null;
}

// The `<details>` block of a bundle's retro: filed on its own by `retro
// --bundle`, or under a task's record by `retro --session`.
function retroBlock(months, id8) {
    const at = months.indexOf('<summary>retro ' + id8 + ' ');
    if (at < 0) return null;
    const end = months.indexOf('</details>', at);
    return months.slice(months.lastIndexOf('<details>', at), end < 0 ? months.length : end + '</details>'.length);
}

// os.homedir() as a transcript spells it: native, with forward slashes, and as
// Git Bash's /c/Users/<name>. Windows paths match in any case; the lookahead
// keeps C:\Users\ann from matching inside C:\Users\anna.
function homePattern(home) {
    const forms = new Set([home, home.split('\\').join('/')]);
    const drive = /^([A-Za-z]):[\\/](.*)$/.exec(home);
    if (drive) forms.add('/' + drive[1].toLowerCase() + '/' + drive[2].split('\\').join('/'));
    const esc = (s) => s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
    const alts = [...forms].sort((a, b) => b.length - a.length).map(esc).join('|');
    return new RegExp('(?:' + alts + ')(?![\\w.-])', process.platform === 'win32' ? 'gi' : 'g');
}

function mapStrings(value, fn) {
    if (typeof value === 'string') return fn(value);
    if (Array.isArray(value)) return value.map((v) => mapStrings(v, fn));
    if (value && typeof value === 'object') {
        const out = {};
        for (const k of Object.keys(value)) out[k] = mapStrings(value[k], fn);
        return out;
    }
    return value;
}

// A bundle as it goes into the ZIP: every string's home directory turned into
// [HOME] unless `re` is null, the head line kept as written, gzip'd again; the
// file in .fankeel/bundles/ is not touched. Also the Claude Code versions its
// entries carry, for the manifest.
function maskBundle(buf, re) {
    const lines = zlib.gunzipSync(buf).toString('utf8').split('\n').filter(Boolean);
    const versions = new Set();
    const out = [lines[0]];
    for (const raw of lines.slice(1)) {
        let row;
        try { row = JSON.parse(raw); } catch (e) { out.push(re ? raw.replace(re, '[HOME]') : raw); continue; }
        if (row && row.e && typeof row.e.version === 'string') versions.add(row.e.version);
        out.push(re ? JSON.stringify(mapStrings(row, (s) => s.replace(re, '[HOME]'))) : raw);
    }
    return { gz: re ? zlib.gzipSync(out.join('\n') + '\n') : buf, versions };
}

// Null for a file that is missing or does not parse, and for a buffer that does not.
function readJson(file) {
    try { return JSON.parse(Buffer.isBuffer(file) ? file.toString('utf8') : fs.readFileSync(file, 'utf8')); } catch (e) { return null; }
}

// Names and versions only. npm exits non-zero over an extraneous package and
// still prints the JSON, so stdout is read from the failure too.
function globalPackages() {
    let text;
    try {
        text = execSync('npm ls -g --depth=0 --json', { encoding: 'utf8', stdio: ['ignore', 'pipe', 'pipe'], timeout: 30000, windowsHide: true });
    } catch (e) {
        text = e && e.stdout;
        if (!text) return 'unavailable: ' + (String((e && (e.stderr || e.message)) || e).trim().split(/\r?\n/)[0] || 'npm failed');
    }
    try {

        const out = {};
        for (const k of Object.keys(deps)) out[k] = deps[k].version || null;
        return out;
    } catch (e) { return 'unavailable: npm printed no JSON'; }
}

    if (!manifest || manifest.export !== 1 || !Array.isArray(manifest.bundles)) fail('Nothing written: ' + zipFile + ' holds no export manifest.json, so it is not a retro export.');
    const bad = [];
    for (const b of manifest.bundles) {
        const data = byName.get('bundles/' + b.name);

}
```
