const TASK_REF = /\btask\s+(\d+)\b/i
export function agentOf(nodes: AgentNode[], n: number): AgentNode | null {
  const all = (list: AgentNode[]): AgentNode[] => list.flatMap((a) => [a, ...all(a.kids)])

// lib/todo.js STATES, the four the TODO block has a Button for, in the station panel's order, each with its label.
export const TODO_STATES: [string, string][] = [['ready', 'Ready'], ['decision', 'Needs a decision'], ['blocked', 'Blocked'], ['watch', 'Watch']]

// The project's open TODO entries under a fold head that counts them, then a
// Button a state with its count, two a row — a state with none is grey words —
// and under them the titles of the state pressed (`open`): a title a row, each a
// Button opening the station's TODO panel on that entry, and past TODO_ROWS a
// grey Button opening the whole panel. The terminal draws a Button four cells
// wider than its label. No entries, no block.
function todoBlock(todos: TodoEntry[], pkey: string, inner: number, folded: boolean, open: string | null): Row[] {
  if (!todos.length) return []
  const at = `#/p/${encodeURIComponent(pkey)}/todo`
  const rows = [row('todo', head('todo', folded, `TODO ${todos.length}`))]
  if (folded) return rows
  const keys = TODO_STATES.map(([state, label]): Seg => {
    const n = todos.filter((t) => t.state === state).length
    if (!n) return dim(`${label} 0`)
    const seg: Seg = { text: `${label} ${n}`, press: 'todoState', state }
    if (state === open) seg.bold = true
    return seg
  })
  rows.push(row('todo', keys[0], { text: ' ' }, keys[1]), row('todo', keys[2], { text: ' ' }, keys[3]))
  const shown = open ? todos.filter((t) => t.state === open) : []
  for (const t of shown.slice(0, TODO_ROWS)) rows.push(row('todo', { text: cut(t.title, Math.max(4, inner - 4)), press: 'todo', to: `${at}/${encodeURIComponent(t.id)}` }))
  if (shown.length > TODO_ROWS) rows.push(row('todo', { text: `還有 ${shown.length - TODO_ROWS} 條`, color: WHERE, press: 'todo', to: at }))
  return rows
}

// docs/90-agent/judgements/2026-10-18-gate-report-fields.md, option C): read out
// of the parsed report's own strings, so the templates in lib/stages.js stay
// inside their 2400-character cap. A copy of `parts` with the additions.
const NONE = /^(none|nothing|無)(?![\w])/i;
const OPENS = '(（[「';
const CLOSES = ')）]」';
// A value's items: split at `；` and `;` always, at `、` only outside brackets
// when `commas`; "none", "nothing", "無" or nothing at all is no items.
function items(text, commas) {
    const s = String(text || '').trim();
    if (!s || NONE.test(s)) return [];
    const out = [];
    let depth = 0;
    let cur = '';
    for (const ch of s) {
        if (OPENS.includes(ch)) depth += 1;
        if (CLOSES.includes(ch)) depth = Math.max(0, depth - 1);
        if (ch === ';' || ch === '；' || (commas && depth === 0 && ch === '、')) {
            if (cur.trim()) out.push(cur.trim());
            cur = '';
            continue;
        }
        cur += ch;
    }
    if (cur.trim()) out.push(cur.trim());
    return out;
}
// The file count a `skipped:` line names: `1528 個檔`, `1528 files`.
const FILES_N = /(\d[\d,]*)\s*個?\s*(?:檔|files?\b)/i;
// The survey lead split at its numbered points, `(1)` or `（1）`: the words before
// the first as the head; fewer than two points is the lead whole, with none.
function pointsOf(lead) {
    const s = String(lead || '');
    const marks = [...s.matchAll(/[(（](\d{1,2})[)）]\s*/g)];
    if (marks.length < 2) return { head: s, points: [] };
    const points = marks.map((m, i) => s.slice(m.index + m[0].length, i + 1 < marks.length ? marks[i + 1].index : undefined).trim());
    return { head: s.slice(0, marks[0].index).trim(), points };
}
// A TODO patrol inside a survey (skills/fankeel-survey/SKILL.md "Every entry,
// one line each"): its `entries: <n> listed, <m> in TODO` line, each entry line's
// disposition — the words after its last colon — counted, and the
// `- do now：…`, `- needs the user：…`, `- waiting on：…` group lines. Null when
// the report has none of it.
const DISPOSITIONS = ['do now', 'needs the user', 'waiting on', 'keep'];
function patrolOf(rest) {
    let listed = null;
    let inTodo = null;
    const counts = { 'do now': 0, 'needs the user': 0, 'waiting on': 0, keep: 0 };
    const groups = [];
    for (const line of rest) {
        const e = /^entries:\s*(\d+)\s+listed(?:,\s*(\d+)\s+in TODO)?/i.exec(line.trim());
        if (e) {
            listed = Number(e[1]);
            inTodo = e[2] ? Number(e[2]) : null;
            continue;
        }
        if (!/^- /.test(line)) continue;
        const g = /^- (do now|needs the user|waiting on)\s*[：:]\s*(.*)$/i.exec(line);
        if (g) {
            groups.push({ name: g[1].toLowerCase(), text: g[2].trim() });
            continue;
        }
        const at = Math.max(line.lastIndexOf('：'), line.lastIndexOf(':'));
        const tail = at >= 0 ? line.slice(at + 1).trim().toLowerCase() : '';
        const d = DISPOSITIONS.find((k) => tail.startsWith(k));
        if (d) counts[d] += 1;
    }
    return listed === null && !Object.values(counts).some(Boolean) ? null : { listed, inTodo, counts, groups };
}
function derive(stage, parts) {
    const out = Object.assign({}, parts);
    if (stage === 'survey') {
        out.lead = pointsOf(parts.conclusion);
        out.unknownList = items(parts.unknown, true);
        out.skippedList = items(parts.skipped, false);

        out.skippedFiles = n ? Number(n[1].replace(/,/g, '')) : null;
        out.patrol = patrolOf(parts.rest || []);
    }
    if (stage === 'design') {
        out.provesList = items(parts.proves, false);
        out.unverifiedList = items(parts.unverified, false);
    }
    return out;
}

    const widths = names.map((n) => /^shot-(\d+)$/.exec(n)).filter(Boolean).map((m) => Number(m[1])).sort((a, b) => b - a);

        if (fs.existsSync(path.join(root, rel))) return { path: rel, href: pathToFileURL(path.join(root, rel)).href };

}

    // The main session's time by stage, from the session row's `stages` (lib/station.js, each stage's
    // first and last touch): a bar a stage as wide as it lasted, a legend under it, and the total.
    function gateTime(s) {
        var segs = (s && s.stages ? s.stages : []).filter(function (w) { return isFinite(w.from) && isFinite(w.to) && w.to > w.from; });
        if (!segs.length) return { total: 0, html: '' };
        var total = segs.reduce(function (n, w) { return n + (w.to - w.from); }, 0);
        var list = segs.map(function (w) { return esc(w.stage) + ' ' + mins(w.to - w.from); }).join(', ');
        var bar = '<div class="gv-bar" role="img" aria-label="' + loc('det.gateTimeAria', '主 session 各站時間：{list}', { list: list }) + '">' + segs.map(function (w) {
            return '<i style="--c:var(--st-' + esc(w.stage) + ');flex:' + Math.round((w.to - w.from) / 1000) + '"></i>';
        }).join('') + '</div>';
        var legend = '<ul class="gv-legend" aria-hidden="true">' + segs.map(function (w) {
            return '<li><i style="--c:var(--st-' + esc(w.stage) + ')"></i>' + esc(w.stage) + ' <b>' + mins(w.to - w.from) + '</b></li>';
        }).join('') + '</ul>';
        return { total: total, html: bar + legend };
    }
    function gateLand(p, s) {
        var ship = p.shipped || [];
        var time = gateTime(s);
        var counts = '<li>' + loc('det.gateNShipped', '<b>{n}</b>件做得到了', { n: ship.length }) + '</li>'
            + (time.total ? '<li>' + loc('det.gateMain', '<b>{t}</b>主 session', { t: mins(time.total) }) + '</li>' : '');
        // lib/gatereport.js runOf: the task's suite.json, its last run. Which files it ran is recorded nowhere.
        var run = p.run;
        var badge = run && run.pass !== null && run.pass !== undefined
            ? '<span class="gv-suite' + (run.ok ? '' : ' gv-red') + '">' + (run.ok ? icon('check') : '') + esc(run.pass) + ' pass · ' + esc(run.fail === null ? '?' : run.fail) + ' fail</span>'
            : '';
        var side = gateSec('suite', null, badge + (p.suite ? '<code class="gv-cmd">' + esc(p.suite) + '</code>' : ''))
            + gateSec(loc('det.gateSecCost', '花了多少'), null, p.cost ? '<div class="gv-split">' + gateText(p.cost) + '</div>' : '')
            + gateSec(loc('det.gateSecTime', '各站時間'), null, time.html)
            + gateSec(loc('det.gateSecOpen', '還沒完'), null, p.open ? '<ul class="gv-open"><li>' + gateText(p.open) + '</li></ul>' : '');
        var body = (p.plain ? '<section class="gv-plain" aria-label="' + loc('det.gatePlain', '白話') + '"><h3>' + loc('det.gatePlain', '白話') + '</h3><p>' + gateText(p.plain) + '</p></section>' : '')
            + (p.sha ? '<p class="gv-commit"><code>' + esc(p.sha) + '</code><span>' + gateText(p.subject) + '</span></p>' : '')
            + '<div class="gv-cols">' + gateSec(loc('det.gateSecShipped', '做得到了'), ship.length, ship.length ? '<ul class="gv-ship">' + ship.map(function (x) {
                return '<li>' + gateText(x) + '</li>';
            }).join('') + '</ul>' : '') + (side ? '<div class="gv-side">' + side + '</div>' : '') + '</div>';
        return { counts: counts, body: body };
    }
