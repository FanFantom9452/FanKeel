# Re-sums the evidence table's own printed columns. Field numbers are the
# columns of `node scripts/spend.js`: 2 root, 3 version, 4 requests, 5 own,
# 6 agents, 8 output, 9 cacheRead, 10 cacheWrite, 11 usd, 12 unpriced.
# ONLY is a regex the root must match; "." keeps every root.
BEGIN { FS = "\t" }
NR > 1 && NF >= 12 && $4 ~ /^[0-9]+$/ && $2 ~ ONLY {
    req = $4 + 0
    b = req < 50 ? "1 <50" : req < 200 ? "2 50-199" : req < 800 ? "3 200-799" : "4 800+"
    own = $5; ag = $6; o = $8; cr = $9; cw = $10; u = $11
    gsub(/\$/, "", own); gsub(/\$/, "", ag); gsub(/\$/, "", o)
    gsub(/\$/, "", cr); gsub(/\$/, "", cw); gsub(/\$/, "", u)
    N[b]++; U[b] += u; M[b] += own; A[b] += ag; O[b] += o; R[b] += cr; W[b] += cw
    tn++; tu += u; tm += own; ta += ag; to += o; tr += cr; tw += cw
}
END {
    printf "%-10s %3s %10s %8s %10s %11s %8s\n", "bucket", "n", "total", "output", "cacheRead", "cacheWrite", "main"
    for (k in U) line[k] = sprintf("%-10s %3d %10.2f %7.1f%% %9.1f%% %10.1f%% %7.1f%%", k, N[k], U[k], 100*O[k]/U[k], 100*R[k]/U[k], 100*W[k]/U[k], 100*M[k]/U[k])
    n = asorti(line, idx)
    for (i = 1; i <= n; i++) print line[idx[i]]
    printf "%-10s %3d %10.2f %7.1f%% %9.1f%% %10.1f%% %7.1f%%\n", "ALL", tn, tu, 100*to/tu, 100*tr/tu, 100*tw/tu, 100*tm/tu
}
